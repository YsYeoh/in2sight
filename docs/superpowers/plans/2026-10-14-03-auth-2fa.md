## Review Focus

1. **Open redirect / redirect loops:** `decideRoute` must never send a signed-in user without finished MFA anywhere but `/login/verify`, never loop between `/login` and `/overview`, and never redirect to a caller-supplied URL. *(Task 1)*
2. **Unauthenticated access to data:** every non-public path (pages, server actions, API routes) is refused without a session; public paths are an explicit allow-list (`/login`, `/setup`, cron and Meta callback endpoints which authenticate themselves). *(Tasks 1-2)*
3. **MFA bypass:** a password-only (aal1) session for an account with a verified factor must reach nothing except `/login/verify`; disabling 2FA must require an aal2 session; abandoned unverified enrolments must not block a new one. *(Tasks 1, 5)*
4. **Secret leakage:** the admin (secret-key) client is unreachable from client components; env errors list variable *names* only. *(Task 1)*
5. **Missing configuration:** with no Supabase env the app must not 500 — it routes to `/setup` and still serves `/styleguide`. *(Tasks 1-2, 7)*
6. **Ownership:** a second Supabase user (sign-ups left on by mistake) can sign in but is not the owner and gets nothing: `is_owner()` guards every policy and the sign-in flow signs them out with `not_owner`. *(Tasks 3-4)*

- [ ] **Step 3: Implement `lib/env.ts`**

```ts
export class EnvError extends Error {
  constructor(public missing: string[]) {
    super(`Missing environment variables: ${missing.join(', ')}. See docs/SETUP.md.`);
    this.name = 'EnvError';
  }
}

const filled = (v: string | undefined): v is string => typeof v === 'string' && v.trim() !== '';

export function isSupabaseConfigured(env: Record<string, string | undefined> = process.env): boolean {
  return filled(env.NEXT_PUBLIC_SUPABASE_URL) && filled(env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY);
}

function need(values: Record<string, string | undefined>): Record<string, string> {
  const missing = Object.entries(values).filter(([, v]) => !filled(v)).map(([k]) => k);
  if (missing.length) throw new EnvError(missing);
  return Object.fromEntries(Object.entries(values).map(([k, v]) => [k, (v as string).trim()]));
}

// NOTE: NEXT_PUBLIC_* must be referenced literally so Next can inline them into the browser bundle.
export function getPublicEnv() {
  const v = need({
    NEXT_PUBLIC_SUPABASE_URL: process.env.NEXT_PUBLIC_SUPABASE_URL,
    NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY: process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY,
    NEXT_PUBLIC_APP_URL: process.env.NEXT_PUBLIC_APP_URL,
  });
  return { supabaseUrl: v.NEXT_PUBLIC_SUPABASE_URL, publishableKey: v.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY, appUrl: v.NEXT_PUBLIC_APP_URL };
}

export function getServerEnv() {
  const v = need({ SUPABASE_SECRET_KEY: process.env.SUPABASE_SECRET_KEY, CRON_SECRET: process.env.CRON_SECRET });
  return { secretKey: v.SUPABASE_SECRET_KEY, cronSecret: v.CRON_SECRET };
}
```

```ts
export interface RouteInput {
  pathname: string;
  configured: boolean;
  signedIn: boolean;
  aal: 'aal1' | 'aal2' | null;
  mfaRequired: boolean;
}
export type RouteDecision = { type: 'next' } | { type: 'redirect'; to: '/login' | '/login/verify' | '/overview' | '/setup' };

const NEXT: RouteDecision = { type: 'next' };
const go = (to: '/login' | '/login/verify' | '/overview' | '/setup'): RouteDecision => ({ type: 'redirect', to });

// Endpoints that authenticate themselves (cron secret / Meta signed_request) and must work without a browser session.
const SELF_AUTHENTICATED = [/^\/api\/cron\/[\w-]+$/, /^\/api\/instagram\/(deauthorize|delete)$/];
const ENTRY = new Set(['/', '/login', '/login/verify', '/setup']);

export function decideRoute({ pathname, configured, signedIn, aal, mfaRequired }: RouteInput): RouteDecision {
  if (!configured) return pathname === '/setup' || pathname === '/styleguide' ? NEXT : go('/setup');
  if (SELF_AUTHENTICATED.some((re) => re.test(pathname))) return NEXT;
  if (!signedIn) return pathname === '/login' ? NEXT : go('/login');

  const needsMfa = mfaRequired && aal !== 'aal2';
  if (needsMfa) return pathname === '/login/verify' ? NEXT : go('/login/verify');
  return ENTRY.has(pathname) ? go('/overview') : NEXT;
}
```

```ts
import type { SupabaseClient } from '@supabase/supabase-js';

export interface SessionInfo {
  signedIn: boolean;
  userId: string | null;
  email: string | null;
  aal: 'aal1' | 'aal2' | null;
  mfaRequired: boolean;
}

const SIGNED_OUT: SessionInfo = { signedIn: false, userId: null, email: null, aal: null, mfaRequired: false };

/** Fails closed: any error means "signed out" / "MFA required". */
export async function getSessionInfo(supabase: SupabaseClient): Promise<SessionInfo> {
  try {
    const { data, error } = await supabase.auth.getClaims();
    const claims = data?.claims as { sub?: string; email?: string; aal?: string } | undefined;
    if (error || !claims?.sub) return SIGNED_OUT;

    const aal = claims.aal === 'aal2' ? 'aal2' : 'aal1';
    let mfaRequired = true;
    if (aal !== 'aal2') {
      const res = await supabase.auth.mfa.getAuthenticatorAssuranceLevel();
      mfaRequired = res.error || !res.data ? true : res.data.nextLevel === 'aal2';
    }
    return { signedIn: true, userId: claims.sub, email: claims.email ?? null, aal, mfaRequired };
  } catch {
    return SIGNED_OUT;
  }
}
```

```ts
import { createServerClient } from '@supabase/ssr';
import { NextResponse, type NextRequest } from 'next/server';
import { decideRoute } from '@/lib/auth/route-policy';
import { getSessionInfo } from '@/lib/auth/session';
import { isSupabaseConfigured } from '@/lib/env';

export async function updateSession(request: NextRequest): Promise<NextResponse> {
  const { pathname } = request.nextUrl;
  const configured = isSupabaseConfigured();

  const redirectTo = (to: string, from?: NextResponse) => {
    const res = NextResponse.redirect(new URL(to, request.url));
    // keep refreshed session cookies (and cache headers) on the redirect
    from?.cookies.getAll().forEach((c) => res.cookies.set(c));
    return res;
  };

  if (!configured) {
    const d = decideRoute({ pathname, configured: false, signedIn: false, aal: null, mfaRequired: false });
    return d.type === 'redirect' ? redirectTo(d.to) : NextResponse.next({ request });
  }

  let response = NextResponse.next({ request });
  const supabase = createServerClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!, {
    cookies: {
      getAll: () => request.cookies.getAll(),
      setAll(list) {
        list.forEach(({ name, value }) => request.cookies.set(name, value));
        response = NextResponse.next({ request });
        list.forEach(({ name, value, options }) => response.cookies.set(name, value, options));
      },
    },
  });

  const info = await getSessionInfo(supabase);
  const d = decideRoute({ pathname, configured: true, signedIn: info.signedIn, aal: info.aal, mfaRequired: info.mfaRequired });
  return d.type === 'redirect' ? redirectTo(d.to, response) : response;
}
```

```sql
-- Single-owner model: exactly one row in public.owners. Every policy in later migrations gates on public.is_owner().

create table public.owners (
  user_id    uuid primary key references auth.users (id) on delete cascade,
  singleton  boolean not null default true unique check (singleton),   -- at most one row, ever
  created_at timestamptz not null default now()
);
alter table public.owners enable row level security;

-- If the owner login already exists when this runs, claim it now (the first/only user).
insert into public.owners (user_id)
select id from auth.users order by created_at asc limit 1
on conflict do nothing;

create or replace function public.is_owner()
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select exists (select 1 from public.owners where user_id = (select auth.uid()))
$$;
revoke all on function public.is_owner() from public;
grant execute on function public.is_owner() to authenticated;

-- Called once after the first successful login if the owners table is still empty. First caller wins.
create or replace function public.claim_ownership()
returns boolean
language plpgsql
security definer
set search_path = public
as $$
declare
  uid uuid := (select auth.uid());
begin
  if uid is null then
    return false;
  end if;
  if exists (select 1 from public.owners) then
    return exists (select 1 from public.owners where user_id = uid);
  end if;
  insert into public.owners (user_id) values (uid) on conflict do nothing;
  return exists (select 1 from public.owners where user_id = uid);
end;
$$;
revoke all on function public.claim_ownership() from public;
grant execute on function public.claim_ownership() to authenticated;

create policy owners_select_self on public.owners
  for select to authenticated
  using (user_id = (select auth.uid()));
```

```ts
import type { SupabaseClient } from '@supabase/supabase-js';

export type SignInError = 'missing_fields' | 'invalid_credentials' | 'not_owner' | 'unexpected';
export type FormState = { error?: SignInError | 'invalid_code' };
export type SignInResult = { ok: true } | { ok: false; error: SignInError };

export async function signInWithPassword(supabase: SupabaseClient, input: { email: string; password: string }): Promise<SignInResult> {
  const email = input.email.trim();
  if (!email || !input.password) return { ok: false, error: 'missing_fields' };
  try {
    const { error } = await supabase.auth.signInWithPassword({ email, password: input.password });
    if (error) return { ok: false, error: 'invalid_credentials' }; // same answer for every failure: no user enumeration

    const { data: isOwner, error: rpcError } = await supabase.rpc('claim_ownership');
    if (rpcError) {
      await supabase.auth.signOut();
      return { ok: false, error: 'unexpected' };
    }
    if (isOwner !== true) {
      await supabase.auth.signOut();
      return { ok: false, error: 'not_owner' };
    }
    return { ok: true };
  } catch {
    return { ok: false, error: 'unexpected' };
  }
```

```ts
import type { SupabaseClient } from '@supabase/supabase-js';

export function normalizeCode(raw: string): string | null {
  const digits = raw.replace(/[\s-]/g, '');
  return /^[0-9]{6}$/.test(digits) ? digits : null;
}

export async function startEnroll(
  supabase: SupabaseClient,
): Promise<{ ok: true; factorId: string; qrSvg: string; secret: string } | { ok: false; error: 'unexpected' }> {
  try {
    // an abandoned, never-verified enrolment would block a new one with the same name
    const { data: existing } = await supabase.auth.mfa.listFactors();
    for (const f of existing?.all ?? []) {
      if (f.status === 'unverified') await supabase.auth.mfa.unenroll({ factorId: f.id });
    }
    const { data, error } = await supabase.auth.mfa.enroll({ factorType: 'totp', friendlyName: 'in2sight' });
    if (error || !data) return { ok: false, error: 'unexpected' };
    return { ok: true, factorId: data.id, qrSvg: data.totp.qr_code, secret: data.totp.secret };
  } catch {
    return { ok: false, error: 'unexpected' };
  }
}

export async function confirmEnroll(supabase: SupabaseClient, input: { factorId: string; code: string }): Promise<{ ok: true } | { ok: false; error: 'invalid_code' | 'unexpected' }> {
  const code = normalizeCode(input.code);
  if (!code) return { ok: false, error: 'invalid_code' };
  try {
    const { error } = await supabase.auth.mfa.challengeAndVerify({ factorId: input.factorId, code });
    return error ? { ok: false, error: 'invalid_code' } : { ok: true };
  } catch {
    return { ok: false, error: 'unexpected' };
  }
}

export async function verifyLogin(supabase: SupabaseClient, rawCode: string): Promise<{ ok: true } | { ok: false; error: 'invalid_code' | 'no_factor' | 'unexpected' }> {
  const code = normalizeCode(rawCode);
  try {
    const { data } = await supabase.auth.mfa.listFactors();
    const factor = data?.totp?.[0];
    if (!factor) return { ok: false, error: 'no_factor' };
    if (!code) return { ok: false, error: 'invalid_code' };
    const { error } = await supabase.auth.mfa.challengeAndVerify({ factorId: factor.id, code });
    return error ? { ok: false, error: 'invalid_code' } : { ok: true };
  } catch {
    return { ok: false, error: 'unexpected' };
  }
}

export async function getMfaStatus(supabase: SupabaseClient): Promise<{ enrolled: boolean; factorId: string | null }> {
  try {
    const { data } = await supabase.auth.mfa.listFactors();
    const factor = data?.totp?.[0];
    return factor ? { enrolled: true, factorId: factor.id } : { enrolled: false, factorId: null };
  } catch {
    return { enrolled: false, factorId: null };
  }
}

export async function disableFactor(supabase: SupabaseClient, factorId: string): Promise<{ ok: true } | { ok: false; error: 'needs_aal2' | 'unexpected' }> {
  try {
    const { data } = await supabase.auth.mfa.getAuthenticatorAssuranceLevel();
    if (data?.currentLevel !== 'aal2') return { ok: false, error: 'needs_aal2' };
    const { error } = await supabase.auth.mfa.unenroll({ factorId });
    return error ? { ok: false, error: 'unexpected' } : { ok: true };
  } catch {
    return { ok: false, error: 'unexpected' };
  }
}
```

```ts
import type { SupabaseClient } from '@supabase/supabase-js';
import { cookies } from 'next/headers';
import { PREFS_COOKIE, parsePrefsCookie, serializePrefs, type Prefs } from '@/lib/prefs';
import { prefsToProfileRow, profileRowToPrefs } from '@/lib/prefs.sync';
import { isSupabaseConfigured } from '@/lib/env';
import { getSessionInfo } from '@/lib/auth/session';
import { createServerSupabase } from '@/lib/supabase/server';

const COOKIE_OPTS = { path: '/', maxAge: 60 * 60 * 24 * 365, sameSite: 'lax' as const };

/** After login: the saved profile wins over whatever this browser had; no profile yet -> create one from the cookie. */
export async function syncPrefsFromProfile(supabase: SupabaseClient): Promise<void> {
  try {
    const info = await getSessionInfo(supabase);
    if (!info.userId) return;
    const jar = await cookies();
    const { data } = await supabase.from('profiles').select('*').eq('user_id', info.userId).maybeSingle();
    const saved = profileRowToPrefs(data);
    if (saved) jar.set(PREFS_COOKIE, serializePrefs(saved), COOKIE_OPTS);
    else await supabase.from('profiles').upsert(prefsToProfileRow(info.userId, parsePrefsCookie(jar.get(PREFS_COOKIE)?.value)));
  } catch {
    // preferences still work from the cookie
  }
}

/** Best effort, never throws: called by setPrefs after the cookie is written. */
export async function persistPrefsToProfile(prefs: Prefs): Promise<void> {
  try {
    if (!isSupabaseConfigured()) return;
    const supabase = await createServerSupabase();
    const info = await getSessionInfo(supabase);
    if (!info.userId) return;
    await supabase.from('profiles').upsert({ ...prefsToProfileRow(info.userId, prefs), updated_at: new Date().toISOString() });
  } catch {
    // keep the cookie value; the next successful save syncs it
  }
}
```
